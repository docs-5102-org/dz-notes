### 1. 安装 Air

### 5. main.go 示例

```go
package main

import (
    "log"
    "os"
    "github.com/gin-gonic/gin"
)

func main() {
    // 根据环境变量设置模式
    env := os.Getenv("APP_ENV")
    if env == "production" {
        gin.SetMode(gin.ReleaseMode)
    } else {
        gin.SetMode(gin.DebugMode)
    }

    router := setupRouter()
    
    port := os.Getenv("PORT")
    if port == "" {
        port = "8080"
    }
    
    log.Printf("Server starting on port %s in %s mode", port, gin.Mode())
    if err := router.Run(":" + port); err != nil {
        log.Fatal("Failed to start server:", err)
    }
}

func setupRouter() *gin.Engine {
    router := gin.Default()
    
    // API v1 分组
    v1 := router.Group("/api/v1")
    {
        v1.GET("/ping", func(c *gin.Context) {
            c.JSON(200, gin.H{
                "message": "pong",
                "version": "1.0.0",
            })
        })
        
        // 用户相关路由
        users := v1.Group("/users")
        {
            users.GET("", getUsers)
            users.GET("/:id", getUser)
            users.POST("", createUser)
            users.PUT("/:id", updateUser)
            users.DELETE("/:id", deleteUser)
        }
    }
    
    return router
}

// Handler 示例
func getUsers(c *gin.Context) {
    c.JSON(200, gin.H{
        "users": []gin.H{
            {"id": 1, "name": "Alice"},
            {"id": 2, "name": "Bob"},
        },
    })
}

func getUser(c *gin.Context) {
    id := c.Param("id")
    c.JSON(200, gin.H{
        "id":   id,
        "name": "User " + id,
    })
}

func createUser(c *gin.Context) {
    var user struct {
        Name  string `json:"name" binding:"required"`
        Email string `json:"email" binding:"required,email"`
    }
    
    if err := c.ShouldBindJSON(&user); err != nil {
        c.JSON(400, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(201, gin.H{
        "message": "User created",
        "user":    user,
    })
}

func updateUser(c *gin.Context) {
    id := c.Param("id")
    c.JSON(200, gin.H{
        "message": "User updated",
        "id":      id,
    })
}

func deleteUser(c *gin.Context) {
    id := c.Param("id")
    c.JSON(200, gin.H{
        "message": "User deleted",
        "id":      id,
    })
}
```

## 方案二：自定义脚本 + Air

### scripts/dev.sh (Linux/Mac)
```bash
#!/bin/bash
export GIN_MODE=debug
export APP_ENV=dev
export PORT=8080
air
```

### scripts/dev.bat (Windows)
```batch
@echo off
set GIN_MODE=debug
set APP_ENV=dev
set PORT=8080
air
```

运行：
```bash
# Linux/Mac
chmod +x scripts/dev.sh
./scripts/dev.sh

# Windows
scripts\dev.bat
```

## 方案三：Makefile（推荐团队使用）

### Makefile
```makefile
.PHONY: dev build run test clean install-tools

# 安装开发工具
install-tools:
	go install github.com/air-verse/air@latest
	go install github.com/golangci/golangci-lint/cmd/golangci-lint@latest

# 开发模式（热重载）
dev:
	air

# 生产构建
build:
	go build -ldflags="-s -w" -o bin/app .

# 运行生产版本
run:
	./bin/app

# 运行测试
test:
	go test -v ./...

# 清理
clean:
	rm -rf tmp/
	rm -rf bin/
	rm -f *.log

# 格式化代码
fmt:
	go fmt ./...
	gofmt -s -w .

# 代码检查
lint:
	golangci-lint run

# 首次设置
setup: install-tools
	go mod download
	air init
```

使用：
```bash
# 首次设置
make setup

# 开发模式
make dev

# 生产构建
make build

# 运行测试
make test
```

## 方案四：使用 Docker + Air（适合团队协作）

### Dockerfile.dev
```dockerfile
FROM golang:1.21-alpine

WORKDIR /app

# 安装 Air
RUN go install github.com/air-verse/air@latest

# 复制依赖文件
COPY go.mod go.sum ./
RUN go mod download

# 复制源代码
COPY . .

# 暴露端口
EXPOSE 8080

# 使用 Air 运行
CMD ["air", "-c", ".air.toml"]
```

### docker-compose.yml
```yaml
version: '3.8'

services:
  app:
    build:
      context: .
      dockerfile: Dockerfile.dev
    ports:
      - "8080:8080"
    volumes:
      - .:/app
      - /app/tmp  # 排除 tmp 目录
    environment:
      - GIN_MODE=debug
      - APP_ENV=dev
```

运行：
```bash
docker-compose up
```

**.air.toml**

### 3. 多环境配置

**config/config.go**
```go
package config

import (
    "os"
)

type Config struct {
    Port     string
    Mode     string
    Database string
}

func Load() *Config {
    return &Config{
        Port:     getEnv("PORT", "8080"),
        Mode:     getEnv("GIN_MODE", "debug"),
        Database: getEnv("DB_URL", "localhost:5432"),
    }
}

func getEnv(key, fallback string) string {
    if value := os.Getenv(key); value != "" {
        return value
    }
    return fallback
}
```

### 4. 优化的项目结构

```go
// routes/routes.go
package routes

import "github.com/gin-gonic/gin"

func SetupRoutes(router *gin.Engine) {
    api := router.Group("/api/v1")
    {
        setupUserRoutes(api)
        setupProductRoutes(api)
    }
}

func setupUserRoutes(rg *gin.RouterGroup) {
    users := rg.Group("/users")

        users.GET("", controllers.GetUsers)
```
