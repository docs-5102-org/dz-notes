## 滑块验证码工作原理

### 1. **缺口检测方法**

**方法一：边缘检测**
- 灰度化 → 高斯模糊 → Canny边缘检测 → 轮廓查找

**方法二：模板匹配**
- 使用滑块作为模板在背景图中匹配

**方法三：像素对比**
- 对比完整图和缺口图的像素差异

### 2. **轨迹生成**

关键是模拟人类行为：
- **加速阶段**：初始加速度较大
- **匀速阶段**：中间保持稳定
- **减速阶段**：接近目标时减速
- **随机抖动**：添加微小的随机偏移

### 3. **现代验证码防御机制**

实际的验证码系统会检测：
- 滑动速度曲线是否自然
- 鼠标轨迹是否符合人类习惯
- 请求频率和设备指纹
- JavaScript环境特征
- 行为生物特征识别

### 4. **合法使用场景**

- 自动化测试自己的应用
- 研究验证码安全性
- 学习计算机视觉技术

import cv2

class SliderCaptchaAnalyzer:
    """滑块验证码分析器 - 教育演示"""
    
    def __init__(self):
        self.background_img = None
        self.slider_img = None
        
    def load_images(self, bg_path, slider_path):
        """加载背景图和滑块图"""
        self.background_img = cv2.imread(bg_path)
        self.slider_img = cv2.imread(slider_path)
        
    def load_images_from_url(self, bg_url, slider_url):
        """从URL加载图片"""
        bg_response = requests.get(bg_url)
        slider_response = requests.get(slider_url)
        
        bg_img = Image.open(BytesIO(bg_response.content))
        slider_img = Image.open(BytesIO(slider_response.content))
        
        self.background_img = cv2.cvtColor(np.array(bg_img), cv2.COLOR_RGB2BGR)
        self.slider_img = cv2.cvtColor(np.array(slider_img), cv2.COLOR_RGB2BGR)
    
    def detect_gap_position(self):
        """
        检测缺口位置
        方法1: 边缘检测
        """
        if self.background_img is None:
            raise ValueError("请先加载图片")
        
        # 转换为灰度图
        bg_gray = cv2.cvtColor(self.background_img, cv2.COLOR_BGR2GRAY)
        
        # 高斯模糊
        bg_blur = cv2.GaussianBlur(bg_gray, (5, 5), 0)
        
        # Canny边缘检测
        edges = cv2.Canny(bg_blur, 100, 200)
        
        # 查找轮廓
        contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        
        # 找到最大的矩形轮廓（可能是缺口）
        max_area = 0
        gap_x = 0
        
        for contour in contours:
            x, y, w, h = cv2.boundingRect(contour)
            area = w * h
            # 缺口通常有特定的宽高比
            if 40 < w < 80 and 40 < h < 80 and area > max_area:
                max_area = area
                gap_x = x
        
        return gap_x
    
    def detect_gap_by_template_matching(self):
        """
        方法2: 模板匹配
        使用滑块作为模板在背景图中查找匹配位置
        """
        if self.background_img is None or self.slider_img is None:
            raise ValueError("请先加载图片")
        
        # 转换为灰度图
        bg_gray = cv2.cvtColor(self.background_img, cv2.COLOR_BGR2GRAY)
        slider_gray = cv2.cvtColor(self.slider_img, cv2.COLOR_BGR2GRAY)
        
        # 模板匹配
        result = cv2.matchTemplate(bg_gray, slider_gray, cv2.TM_CCOEFF_NORMED)
        
        # 找到最佳匹配位置
        min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)
        
        return max_loc[0]

    def generate_track(self, distance):
        """
        生成滑动轨迹
        模拟人类滑动行为：先加速后减速
        
        Args:
            distance: 需要滑动的距离（像素）
        
        Returns:
            轨迹列表 [(x, y, t), ...]
        """
        track = []
        current = 0
        mid = distance * 4 / 5  # 加速到4/5处开始减速
        t = 0.2
        v = 0
        
        while current < distance:
            if current < mid:
                # 加速阶段
                a = 2
            else:
                # 减速阶段
                a = -3
            
            v0 = v
            v = v0 + a * t
            move = v0 * t + 1/2 * a * t * t
            current += move
            
            # 添加随机扰动，模拟人类操作
            track.append({
                'x': round(current + np.random.uniform(-1, 1)),
                'y': round(np.random.uniform(-1, 1)),
                'time': round(t * 1000)
            })
        
        return track


class SliderCaptchaSimulator:
    """滑块验证码模拟器 - 用于测试"""
    
    @staticmethod
    def create_captcha_image(width=300, height=150):
        """创建一个简单的验证码图片"""
        # 创建背景图
        bg = np.ones((height, width, 3), dtype=np.uint8) * 240
        
        # 添加一些纹理
        for _ in range(100):
            x, y = np.random.randint(0, width), np.random.randint(0, height)
            cv2.circle(bg, (x, y), 2, (200, 200, 200), -1)
        
        # 创建缺口位置
        gap_x = np.random.randint(100, 200)
        gap_y = 50
        gap_width = 50
        gap_height = 50
        
        # 在背景图上绘制缺口轮廓
        cv2.rectangle(bg, (gap_x, gap_y), 
                     (gap_x + gap_width, gap_y + gap_height), 
                     (100, 100, 100), 2)
        
        # 创建滑块图
        slider = np.ones((gap_height + 20, gap_width + 20, 3), dtype=np.uint8) * 255
        cv2.rectangle(slider, (10, 10), 
                     (gap_width + 10, gap_height + 10), 
                     (100, 100, 100), -1)
        
        return bg, slider, gap_x
